using System;
using System.Collections.Generic;
using System.Globalization;
using System.Text;
using NUnit.Framework;
using Planetwalker.Core;
using Planetwalker.Generation;
using Planetwalker.Persistence;
using Planetwalker.Rendering;
using UnityEngine;

namespace Planetwalker.Tests
{
    /// <summary>G.2D reconnaissance: ribbon sample counts, vertical offsets vs the canonical surface, horizontal offsets vs the carved axes, costs.</summary>
    public class G2DDiag
    {
        static readonly CultureInfo Inv = CultureInfo.InvariantCulture;

        [Test]
        public void Run()
        {
            var log = new StringBuilder();
            var c = new WorldConfiguration();
            c.Hydrology.riverNetworkThreshold = 1e-5f; // River Density 100 % in the panel
            PlanetDefinition def = c.Terrain.Clone(); def.seed = 1337; def.erosionStrength = 1f; def.erosionModel = 1; def.regionalLandforms = 1f; def.surfaceDetail = true;
            var planet = new PlanetInstance(def);
            double R = planet.Scale.RadiusMeters;
            ValleyNetwork net = planet.Height.SurfaceDetail.Landforms.Valleys;

            // The river network as the visual controller builds it (legacy hydrology at the archived resolution).
            var hydrology = new HydrologySystem().Compute(new HydrologyGrid(c.Hydrology.resolution), planet);
            var sw = System.Diagnostics.Stopwatch.StartNew();
            var system = new RiverNetworkSystem();
            RiverNetwork network = system.Build(hydrology, c.Hydrology.riverNetworkThreshold);
            system.BuildPaths(network, c.Rivers.SmoothingIterations);
            double networkMs = sw.Elapsed.TotalMilliseconds;
            log.AppendLine(string.Format(Inv, "hydrology {0} ({1} cells), rivers {2}, path points {3}, network build {4:0} ms", c.Hydrology.resolution, hydrology.Grid.CellCount, network.RiverCount, network.PathPointCount, networkMs));

            // Legacy surface sampler (face meshes at the archived resolution) vs the canonical surface along the path points.
            var faces = new Mesh[CubeSphereGenerator.FaceCount];
            var meshBuilder = new PlanetMeshBuilder();
            for (int f = 0; f < faces.Length; f++) { faces[f] = new Mesh(); meshBuilder.BuildFace(faces[f], f, planet, PlanetColorRamp.CreateDefault()); }
            var sampler = new PlanetSurfaceSampler();
            sampler.Capture(faces, def.resolution);
            int n = 0, inRange = 0, floating = 0, buried = 0; double sumAbs = 0, maxAbove = 0, maxBelow = 0, sumH = 0; int nH = 0; double maxH = 0;
            var vertical = new List<float>(); var horizontal = new List<float>();
            sw.Restart();
            double canonicalUs = 0;
            for (int i = 0; i < network.PathPointCount; i += 25)
            {
                Vector3 d = network.GetPathPoint(i).normalized;
                float legacy = sampler.Sample(d).magnitude;
                var t0 = System.Diagnostics.Stopwatch.GetTimestamp();
                float canonical = planet.Radius * (1f + planet.Height.GetElevation(new Vector3d(d)));
                canonicalUs += (System.Diagnostics.Stopwatch.GetTimestamp() - t0) * 1e6 / System.Diagnostics.Stopwatch.Frequency;
                double dv = (legacy - canonical) / planet.Radius * R;
                n++; sumAbs += Math.Abs(dv); vertical.Add((float)dv);
                if (dv > 20) floating++; if (dv < -20) buried++;
                maxAbove = Math.Max(maxAbove, dv); maxBelow = Math.Min(maxBelow, dv);
                // Horizontal: distance from the river point to the carved axis (terrain space: the meander inverted).
                planet.Height.SampleLayers(d, out TerrainDetailSample s);
                if (s.Landforms.Valley > 0.3f && net.Nearest(new Vector3d(d), out ValleyNetwork.ChannelQuery q))
                {
                    inRange++;
                    horizontal.Add(q.DistanceMeters); sumH += q.DistanceMeters; nH++; maxH = Math.Max(maxH, q.DistanceMeters);
                }
            }
            vertical.Sort(); horizontal.Sort();
            log.AppendLine(string.Format(Inv, "path points sampled {0}: legacy minus canonical: mean |dv| {1:0} m, p50 {2:0} p90 {3:0} max above {4:0} m (floating), max below {5:0} m (buried); floating (>20 m) {6}, buried {7}",
                n, sumAbs / n, vertical[vertical.Count / 2], vertical[(int)(vertical.Count * 0.9)], maxAbove, maxBelow, floating, buried));
            log.AppendLine(string.Format(Inv, "in the ranges (valley region > 0.3): {0} points; distance to the nearest carved axis (meander space): mean {1:0} m, p50 {2:0} p90 {3:0} max {4:0} m",
                inRange, nH > 0 ? sumH / nH : 0, nH > 0 ? horizontal[nH / 2] : 0, nH > 0 ? horizontal[(int)(nH * 0.9)] : 0, maxH));
            log.AppendLine(string.Format(Inv, "canonical height cost {0:0.0} us/sample (sequential)", canonicalUs / n));
            // Total displayed path length and the sample counts at spacings.
            double length = 0, lengthRange = 0;
            for (int r = 0; r < network.RiverCount; r++)
            {
                int s0 = network.GetPathStart(r), s1 = network.GetPathEnd(r);
                for (int i = s0; i + 1 < s1; i++)
                {
                    double l = (network.GetPathPoint(i + 1) - network.GetPathPoint(i)).magnitude * R;
                    length += l;
                    planet.Height.SampleLayers(network.GetPathPoint(i).normalized, out TerrainDetailSample s);
                    if (s.Landforms.Valley > 0.3f) lengthRange += l;
                }
            }
            log.AppendLine(string.Format(Inv, "total path length {0:0} km ({1:0} km in the ranges); samples at 39 km: {2:0}k, at 5 km: {3:0}k, at 2 km: {4:0}k, at 1 km: {5:0}k",
                length / 1000, lengthRange / 1000, length / 39000 / 1000, length / 5000 / 1000, length / 2000 / 1000, length / 1000 / 1000));
            foreach (Mesh m in faces) UnityEngine.Object.DestroyImmediate(m);
            System.IO.File.WriteAllText(@"C:\Users\Philippe\AppData\Local\Temp\claude\C--Users-Philippe-Documents-Programming-Unity-Projects-planetwalker\92a57fea-0dca-4c6f-a05a-b46236670522\scratchpad\g2ddiag.txt", log.ToString());
        }
    }
}
