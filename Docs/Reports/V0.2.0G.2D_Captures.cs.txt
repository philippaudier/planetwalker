        static GameObject Shell(PlanetVisualController visual, float radius, Mesh mesh, Material material, string blockField)
        {
            var shell = new GameObject("Shell");
            shell.transform.SetParent(visual.transform, false);
            shell.transform.localScale = Vector3.one * radius;

            var mr = shell.AddComponent<MeshRenderer>();

        void SaveDiff(Color32[] a, Color32[] b, string name, string scenario, string note)
        {
            var d = new Color32[a.Length];
            for (int i = 0; i < a.Length; i++)
            {
                int v = Math.Min(255, 8 * (Math.Abs(a[i].r - b[i].r) + Math.Abs(a[i].g - b[i].g) + Math.Abs(a[i].b - b[i].b)) / 3);
                d[i] = new Color32((byte)v, (byte)v, (byte)v, 255);
            }
            Save(d, name, scenario, note + " (absolute difference x8)");
        }

        static double Diff(Color32[] a, Color32[] b)
        {
            long sum = 0;
            for (int i = 0; i < a.Length; i++) sum += Math.Abs(a[i].r - b[i].r) + Math.Abs(a[i].g - b[i].g) + Math.Abs(a[i].b - b[i].b);
            return sum / (3.0 * a.Length);
        }

        static double Changed(Color32[] a, Color32[] b, int threshold = 6)
        {
            long n = 0;
            for (int i = 0; i < a.Length; i++) if (Math.Abs(a[i].r - b[i].r) + Math.Abs(a[i].g - b[i].g) + Math.Abs(a[i].b - b[i].b) > threshold) n++;
            return 100.0 * n / a.Length;
        }

        static double Luma(Color32[] f)
        {
            double s = 0; int n = 0;
            for (int y = 0; y < H / 2; y++) for (int x = W / 4; x < 3 * W / 4; x++) { Color32 q = f[x + y * W]; s += 0.3 * q.r + 0.59 * q.g + 0.11 * q.b; n++; }
            return s / n;
        }

        void Settle()

        void Pose(Vector3d at, double meters, Vector3d facing, double pitchDeg)
        {
            Vector3d u = at.Normalized;
            Vector3d f = (facing - u * Vector3d.Dot(facing, u)).Normalized;
            double p = pitchDeg * Math.PI / 180.0;
            explorer.SetPose(u * (explorer.FloorRadius(u) + scale.ToUnits(meters)), f * Math.Cos(p) + u * Math.Sin(p), f);
            explorer.Place();
        }

        float SunHeight(Vector3d d)

            return Vector3.Dot(d.Normalized.ToVector3(), sun);
        }

        void LightPoint(Vector3d d, float height)
        {
            float best = 0.42f, bestErr = float.MaxValue;
            for (int k = 0; k < 400; k++)

                float t = k / 400f;
                host.Visual.SetSolarTime(t);
                float err = Mathf.Abs(SunHeight(d) - height) + (t > 0.5f ? 0.001f : 0f);
                if (err < bestErr) { bestErr = err; best = t; }
            }
            host.Visual.SetSolarTime(best);
        }

        /// <summary>Background pixels enclosed by terrain in their column (terrain above and below): holes.</summary>
        static int Holes(Color32[] px)
        {
            int holes = 0;
            for (int x = 0; x < W; x++)
            {
                bool terrainAbove = false;
                int run = 0;
                for (int y = H - 1; y >= 0; y--)
                {
                    Color32 q = px[x + y * W];
                    bool background = q.r > 240 && q.g < 20 && q.b > 240;
                    if (!background)
                    {
                        if (terrainAbove) holes += run;
                        terrainAbove = true;
                        run = 0;
                    }
                    else if (terrainAbove)
                    {
                        run++;
                    }
                }
            }
            return holes;
        }

        struct Hit { public double Distance; public bool Skirt; }

        bool Raycast(Vector3 origin, Vector3 dir, out Hit best)
        {
            best = new Hit { Distance = double.MaxValue };
            var cache = (PatchMeshCache)typeof(AdaptiveTerrainRenderer).GetField("meshes", Private).GetValue(terrain);
            int grid = terrain.Snapshot.Topology.GridVertexCount;
            foreach (PlanetPatchId leaf in terrain.Displayed.Leaves)
            {
                if (!cache.TryGet(leaf, out PatchMeshCache.Entry entry)) continue;
                Matrix4x4 m = terrain.PatchMatrix(entry.Geometry.Origin);
                Bounds b = entry.Geometry.LocalBounds;
                Vector3 c = m.MultiplyPoint3x4(b.center);
                float radius = b.extents.magnitude * Math.Abs(host.Root.transform.lossyScale.x);
                Vector3 oc = c - origin;
                float along = Vector3.Dot(oc, dir);
                if ((oc - dir * along).sqrMagnitude > radius * radius * 1.01f && oc.sqrMagnitude > radius * radius) continue;
                Vector3[] v = entry.Mesh.vertices;
                int[] t = entry.Mesh.GetIndices(0);
                for (int i = 0; i < t.Length; i += 3)
                {
                    Vector3 a = m.MultiplyPoint3x4(v[t[i]]), bb = m.MultiplyPoint3x4(v[t[i + 1]]), cc = m.MultiplyPoint3x4(v[t[i + 2]]);
                    if (Intersect(origin, dir, a, bb, cc, out double d) && d < best.Distance)
                    {
                        best = new Hit { Distance = d, Skirt = t[i] >= grid || t[i + 1] >= grid || t[i + 2] >= grid };
                    }
                }
            }
            return best.Distance < double.MaxValue;
        }

        static bool Intersect(Vector3 o, Vector3 d, Vector3 a, Vector3 b, Vector3 c, out double t)
        {
            double ex1 = b.x - a.x, ey1 = b.y - a.y, ez1 = b.z - a.z, ex2 = c.x - a.x, ey2 = c.y - a.y, ez2 = c.z - a.z;
            double px = d.y * ez2 - d.z * ey2, py = d.z * ex2 - d.x * ez2, pz = d.x * ey2 - d.y * ex2;
            double det = ex1 * px + ey1 * py + ez1 * pz;
            t = 0;
            if (Math.Abs(det) < 1e-24) return false;
            double inv = 1.0 / det, sx = o.x - a.x, sy = o.y - a.y, sz = o.z - a.z;
            double u = (sx * px + sy * py + sz * pz) * inv;
            if (u < 0 || u > 1) return false;
            double qx = sy * ez1 - sz * ey1, qy = sz * ex1 - sx * ez1, qz = sx * ey1 - sy * ex1;
            double v = (d.x * qx + d.y * qy + d.z * qz) * inv;
            if (v < 0 || u + v > 1) return false;
            t = (ex2 * qx + ey2 * qy + ez2 * qz) * inv;
            return t > 0;
        }

        /// <summary>Distance from the camera to the rendered mesh (rays over the sphere), render units.</summary>
        double NearestMesh()
        {
            double best = double.MaxValue;
            const int rays = 300;
            Vector3 o = camera.transform.position;
            for (int q = 0; q < rays; q++)
            {
                double yy = 1.0 - 2.0 * (q + 0.5) / rays, rr = Math.Sqrt(1.0 - yy * yy), ph = q * 2.399963229728653;
                var dir = new Vector3((float)(rr * Math.Cos(ph)), (float)yy, (float)(rr * Math.Sin(ph)));
                if (Raycast(o, dir, out Hit h) && h.Distance < best) best = h.Distance;
            }
            return best;
        }

        /// <summary>|rendered mesh - authoritative height| on a 10 x 10 grid ahead of a point (50 .. 500 m), meters.</summary>
        (double max, double mean) ApproximationError(Vector3d point, Vector3d f)
        {
            PlanetRenderFrame frame = host.Root.GetComponent<PlanetRenderFrame>();
            Vector3d up = point.Normalized;
            Vector3d east = Vector3d.Cross(f, up).Normalized;
            double toM = scale.MetersPerUnit / Math.Abs(host.Root.transform.lossyScale.x);
            double max = 0, sum = 0; int count = 0;
            for (int i = 0; i < 10; i++)
            for (int j = 0; j < 10; j++)
            {
                double ahead = 50.0 + i * 50.0, side = (j - 4.5) * 40.0;
                Vector3d d = (up + f * (ahead / radiusM) + east * (side / radiusM)).Normalized;
                double floor = explorer.FloorRadius(d);
                Vector3 top = frame.PlanetToRender(d * (floor + scale.ToUnits(3000.0)));
                Vector3 down = -frame.PlanetVectorToRender(d).ToVector3().normalized;
                if (!Raycast(top, down, out Hit h)) continue;
                double meshAboveFloor = 3000.0 - h.Distance * toM;
                // Only land (the sea surface is the floor over water; the mesh is the sea bottom there).
                if (planet.Height.GetElevation(d) <= planet.SeaLevel) continue;
                double e = Math.Abs(meshAboveFloor);
                max = Math.Max(max, e); sum += e; count++;
            }
            return (max, count > 0 ? sum / count : 0);
        }

        public void Run()
        {

            c.Rivers.placement = RiverSettings.PlacementCanonical;

            host.Definition.seed = Seed;
            host.Definition.surfaceDetail = true;
            host.Definition.regionalLandforms = 1f; host.Definition.erosionStrength = 1f; host.Definition.erosionModel = 1;
            host.Visual.Climate.resolution = 64;
            host.Regenerate();
            PlanetRenderFrame frame = PlanetRenderFrame.GetOrCreate(host.Root.transform);
            foreach (Light l in Object.FindObjectsByType<Light>(FindObjectsSortMode.None)) { if (l.name == "Archive Test Star") RenderSettings.sun = l; else l.enabled = false; }
            explorer = host.Camera.gameObject.AddComponent<PlanetExploreCamera>();
            host.Camera.enabled = false;
            explorer.Target = host.Root.transform;
            explorer.Planet = host.Current;
            camera = host.Camera.GetComponent<Camera>();
            terrain = AdaptiveTerrainRenderer.GetOrCreate(host.Visual);
            terrain.BackgroundConstruction = false;
            terrain.ConstructionMilliseconds = 100000f;
            terrain.MaxUploadsPerFrame = 4096;
            terrain.MaxPendingBuilds = 4096;
            terrain.ViewCamera = camera;
            terrain.enabled = true;
            terrain.MaterialMode = SurfaceMaterialMode.Pbr;
            terrain.Tick(camera);
            planet = host.Current;
            scale = planet.Scale;
            radiusM = scale.RadiusMeters;
            PlanetVisualController visual = host.Visual;
            // Rivers need the hydrology: force the river update once (RiversNeeded is true with showRivers).
            visual.RefreshColors();
            RiverNetwork network = visual.CurrentRivers;
            RiverRibbonBuilder spine = visual.RiverRibbons;
            RiverSurface surface = visual.RiverSurface;
            Assert.IsTrue(surface.HasSurface, "canonical surface built");
            // The prepass runs on workers in the controller: finish it here before reading the placed spine.
            visual.TickRiverSurface(camera, true);
            ValleyNetwork net = planet.Height.SurfaceDetail.Landforms.Valleys;
            log.Append(string.Format(Inv, "Default configuration, seed {0}, hydrology {1}, threshold {2}: {3} rivers, {4} ribbons, {5} spine samples, {6} pieces; prepass {7:0} ms CPU",
                Seed, visual.Hydrology.resolution, visual.Hydrology.riverNetworkThreshold, network.RiverCount, spine.RibbonCount, spine.SpineSampleCount, surface.PieceCount, surface.Diagnostics.PrepassMilliseconds));
            RiverSurfaceDiagnostics d0 = surface.Diagnostics;
            log.Append(string.Format(Inv, "\nCorridors: {0} / {1} trajectory nodes on carved cells, links matched {2} / mismatched {3}; snapped {4} / {5} spine samples, offset mean {6:0} m max {7:0} m, residual max {8:0.000} m; mouths above the sea on the placed surface {9}",
                d0.ChannelNodes, d0.TrajectoryNodes, d0.MatchedLinks, d0.MismatchedLinks, d0.SnappedSamples, d0.SpineSamples, d0.MeanOffsetMeters, d0.MaxOffsetMeters, d0.MaxResidualMeters, d0.CanonicalMouthGaps));

            // ---- Locations, from the displayed rivers on this planet (not from the G.2B.1 test-configuration presets).
            Vector3d Dir(int i) { surface.GetSpineSample(i, out Vector3d dd, out _, out _, out _, out _); return dd; }
            float Weight(int i) { surface.GetSpineSample(i, out _, out _, out float w, out _, out _); return w; }
            float Incision(Vector3d dd) => net.Nearest(dd, out ValleyNetwork.ChannelQuery q) ? q.IncisionMeters : 0f;
            Vector3d Flow(Vector3d dd) => net.Nearest(dd, out ValleyNetwork.ChannelQuery q) && q.Flow.Magnitude > 0 ? q.Flow : PlanetTangentFrame.At(dd).North;
            int ribbons = surface.RibbonCount;
            int gorge = -1, head = -1, confluence = -1, floor = -1, mouth = -1;
            float gorgeI = -1f, headI = -1f, confluenceI = -1f, floorScore = -1f; double mouthArea = -1;
            int nSnapped = 0, nNear = 0, nEnds = 0, nTribEnds = 0, nBroad = 0; float maxCells = 0f;
            for (int r = 0; r < ribbons; r++)
            {
                // Surface-side ranges (confluence samples appended): never the builder's indices.
                int a = surface.GetRibbonStart(r), b = surface.GetRibbonStart(r + 1);
                if (b - a < 2) continue;
                River data = network.GetRiver(surface.GetSpineRiver(a));
                for (int i = a; i < b; i++)
                {
                    Vector3d dd = Dir(i);
                    if (Math.Abs(dd.y) > 0.75) continue;
                    float w = Weight(i);
                    if (w < 0.999f) continue;
                    nSnapped++;
                    if (!net.Nearest(dd, out ValleyNetwork.ChannelQuery q)) continue;
                    nNear++;
                    if (i == b - 1) { nEnds++; if (data.ParentRiver >= 0) nTribEnds++; }
                    maxCells = Math.Max(maxCells, q.Cells);
                    if (q.Cells >= 16f) nBroad++;
                    if (q.IncisionMeters > gorgeI) { gorgeI = q.IncisionMeters; gorge = i; }
                    // Channel head: where a source river enters its carved corridor (the legacy source itself drains too
                    // little for a carved valley, which starts at 2 cells): the first snapped sample of a source ribbon.
                    if (data.ParentRiver < 0 && (i == a || Weight(i - 1) < 0.999f) && q.IncisionMeters > headI) { headI = q.IncisionMeters; head = i; }
                    if (i == b - 1 && data.ParentRiver >= 0 && q.IncisionMeters > confluenceI) { confluenceI = q.IncisionMeters; confluence = i; }
                    float broad = q.Cells * (q.HalfWidthMeters / 11000f) / (1f + q.IncisionMeters / 300f);
                    if (q.Cells >= 16f && broad > floorScore) { floorScore = broad; floor = i; }
                }
                if (data.MouthCell >= 0 && data.DrainedArea > mouthArea && Math.Abs(Dir(b - 1).y) < 0.75)
                {
                    // A mouth whose river crosses the ranges (some snapped sample upstream); the camera stands on the
                    // last sample still above the sea (the ribbon continues under the rendered ocean).
                    bool carved = false;
                    for (int i = a; i < b && !carved; i++) carved = Weight(i) > 0.5f;
                    int dry = -1;
                    for (int i = b - 1; i > a && dry < 0; i--) { surface.GetSpineSample(i, out _, out _, out _, out _, out float e0); if (e0 > planet.SeaLevel) dry = i; }
                    if (carved && dry > a) { mouthArea = data.DrainedArea; mouth = dry; }
                }
            }
            log.Append(string.Format(Inv, "\nLocation samples: head {0}, gorge {1}, confluence {2}, valley floor {3}, outlet {4} (-1 = none found); candidates: snapped {5}, with a nearest channel {6}, ribbon ends {7}, tributary ends {8}, with >= 16 cells {9}, max cells {10:0}", head, gorge, confluence, floor, mouth, nSnapped, nNear, nEnds, nTribEnds, nBroad, maxCells));
            Assert.IsTrue(gorge >= 0, "a gorge location exists");
            var presets = new List<(string name, int sample, (double, double, string)[] alts)>();
            if (head >= 0) presets.Add(("head", head, new[] { (20000.0, -35.0, "20km"), (2000.0, -15.0, "2km"), (200.0, -6.0, "200m"), (20.0, -3.0, "20m") }));
            presets.Add(("gorge", gorge, new[] { (200000.0, -60.0, "200km"), (20000.0, -35.0, "20km"), (2000.0, -15.0, "2km"), (200.0, -6.0, "200m"), (20.0, -3.0, "20m") }));
            if (confluence >= 0) presets.Add(("confluence", confluence, new[] { (20000.0, -35.0, "20km"), (2000.0, -15.0, "2km"), (200.0, -6.0, "200m"), (20.0, -3.0, "20m") }));
            if (floor >= 0) presets.Add(("valley_floor", floor, new[] { (20000.0, -35.0, "20km"), (2000.0, -15.0, "2km"), (200.0, -6.0, "200m"), (20.0, -3.0, "20m") }));
            if (mouth >= 0) presets.Add(("outlet", mouth, new[] { (20000.0, -35.0, "20km"), (2000.0, -15.0, "2km"), (200.0, -6.0, "200m") }));
            foreach ((string name, int sample, _) in presets)
            {
                Vector3d dd = Dir(sample);
                RiverSpineSample legacy = spine.GetSpineSample(sample);
                surface.GetSpineSample(sample, out _, out float hw, out float w, out float offset, out float e0);
                log.Append(string.Format(Inv, "\n{0}: at ({1:0.000000}, {2:0.000000}, {3:0.000000}), corridor weight {4:0.00}, legacy offset {5:0} m, incision {6:0} m, half width {7:0} m, level-0 elevation {8:0} m",
                    name, dd.x, dd.y, dd.z, w, offset, Incision(dd), hw * radiusM, (e0 - planet.SeaLevel) * radiusM));
            }

            // Numeric probe of each location: the terrain under the camera, the placed surface at three levels, and after a
            // 20 m pose what the near piece under the camera actually displays (level, vertex radius vs the terrain).
            foreach ((string name, int sample, _) in presets)
            {
                Vector3d dd = Dir(sample);
                double full = (planet.Height.GetElevation(dd) - planet.SeaLevel) * radiusM;
                double s0 = (planet.Height.GetSurfaceElevation(dd, surface.GetLevelStep(0)) - planet.SeaLevel) * radiusM;
                double s4 = (planet.Height.GetSurfaceElevation(dd, surface.GetLevelStep(4)) - planet.SeaLevel) * radiusM;
                double s11 = (planet.Height.GetSurfaceElevation(dd, surface.GetLevelStep(11)) - planet.SeaLevel) * radiusM;
                double floorM = (explorer.FloorRadius(dd) / planet.Radius - 1.0 - planet.SeaLevel) * radiusM;
                Pose(dd, 20.0, Flow(dd), -3.0);
                SettleRivers();
                int piece = -1;
                for (int p = 0; p < surface.PieceCount; p++) if (surface.GetPieceFirstSample(p) == sample) piece = p;
                string shown = "no piece";
                if (piece >= 0 && surface.GetPieceVertices(piece, out int group, out int first, out int count))
                {
                    Mesh m = surface.GetGroupMesh(group, piece, out Vector3 origin);
                    Vector3[] v = m.vertices;
                    double rl = (origin + v[first]).magnitude, rr = (origin + v[first + 1]).magnitude;
                    double hl = (rl / planet.Radius - 1.0 - planet.SeaLevel) * radiusM, hr = (rr / planet.Radius - 1.0 - planet.SeaLevel) * radiusM;
                    shown = string.Format(Inv, "piece {0} level {1} group {2}: first left / right vertex {3:0.0} / {4:0.0} m", piece, surface.GetDisplayedLevel(piece), group, hl, hr);
                }
                log.Append(string.Format(Inv, "\nProbe {0}: terrain {1:0.0} m, surface L0 {2:0.0} L4 {3:0.0} L11 {4:0.0} m, explorer floor {5:0.0} m, camera {6:0.0} m above the floor; {7}",
                    name, full, s0, s4, s11, floorM, explorer.ClearanceMeters, shown));
                // Every piece whose displayed geometry comes within 5 km of the camera: level, group, weight, and how high
                // its vertices rise above the camera (a coarse chord flying over the viewpoint shows as a sheet above).
                Vector3d cam = explorer.Position;
                int listed = 0;
                var meshVertices = new Dictionary<Mesh, Vector3[]>();
                for (int p = 0; p < surface.PieceCount && listed < 12; p++)
                {
                    if (!surface.GetPieceVertices(p, out int grp, out int vFirst, out int vCount)) continue;
                    Mesh pm = surface.GetGroupMesh(grp, p, out Vector3 pOrigin);
                    if (pm == null) continue;
                    if (!meshVertices.TryGetValue(pm, out Vector3[] pv)) meshVertices[pm] = pv = pm.vertices;
                    double nearest = double.MaxValue, highest = double.MinValue;
                    for (int k = vFirst; k < vFirst + vCount && k < pv.Length; k++)
                    {
                        Vector3d w = new Vector3d(pOrigin + pv[k]);
                        nearest = Math.Min(nearest, (w - cam).Magnitude * radiusM / planet.Radius);
                        highest = Math.Max(highest, (w.Magnitude - cam.Magnitude) * radiusM / planet.Radius);
                    }
                    if (nearest > 5000.0) continue;
                    surface.GetSpineSample(surface.GetPieceFirstSample(p), out _, out float phw, out float pw, out _, out _);
                    log.Append(string.Format(Inv, "\n   near piece {0}: level {1}, group {2}, river {3}, weight {4:0.00}, half width {5:0} m, nearest vertex {6:0} m, highest vertex {7:+0;-0} m relative to the camera",
                        p, surface.GetDisplayedLevel(p), grp, surface.GetSpineRiver(surface.GetPieceFirstSample(p)), pw, phw * radiusM, nearest, highest));
                    listed++;
                }
            }

            var configs = new[] { ("A", RiverSettings.PlacementLegacy), ("B", RiverSettings.PlacementCanonical) };
            var frames = new Dictionary<string, Color32[]>();
            foreach ((string config, int placement) in configs)
            {
                visual.Rivers.placement = placement;
                visual.RefreshColors(); // rebuilds the ribbons for the placement
                foreach ((string name, int sample, (double, double, string)[] alts) in presets)
                {
                    Vector3d at = Dir(sample);
                    Vector3d facing = name == "outlet" ? (Dir(sample + 1) - Dir(sample - 1)).Normalized : Flow(at);
                    LightPoint(at, 0.55f);
                    foreach ((PlanetDebugView view, string viewName) in new[] { (PlanetDebugView.None, "pbr"), (PlanetDebugView.RiverAlignment, "align") })
                    {
                        visual.DebugView = view;
                        foreach ((double meters, double pitch, string tag) in alts)
                        {
                            Pose(at, meters, facing, pitch);
                            SettleRivers();
                            Color32[] img = Render();
                            string key = $"{name}_{tag}_{viewName}";
                            frames[config + "_" + key] = img;
                            RiverSurfaceDiagnostics d = visual.RiverSurface.Diagnostics;
                            string note = $"seed {Seed} {name}, {tag}, pitch {pitch}, view {viewName}, placement {(placement == 1 ? "canonical" : "legacy")}, sun height {SunHeight(at):0.00}";
                            if (placement == 1) note += $", surface samples {d.DisplayedSamples} (far {d.FarPieces} mid {d.MidPieces} near {d.NearPieces}), tick {visual.RiverSurfaceTickMilliseconds:0.0} ms";
                            Save(img, $"{config}_{key}", "G.2D " + config, note);
                            if (config == "B" && frames.TryGetValue("A_" + key, out Color32[] before))
                            {
                                SaveDiff(before, img, $"D_{key}", "G.2D A-B", $"{name} {tag} {viewName}: changed pixels {Changed(before, img):0.0} %");
                            }
                        }
                    }
                    visual.DebugView = PlanetDebugView.None;
                }
            }

            // ---- LOD descent / ascent at the gorge (B): the same pose seen on the way down and on the way up.
            {
                visual.Rivers.placement = RiverSettings.PlacementCanonical;
                visual.RefreshColors();
                visual.DebugView = PlanetDebugView.None;
                Vector3d at = Dir(gorge);
                Vector3d facing = Flow(at);
                LightPoint(at, 0.55f);
                var series = new[] { (50000.0, -40.0, "50km"), (5000.0, -20.0, "5km"), (500.0, -8.0, "500m"), (50.0, -4.0, "50m"), (500.0, -8.0, "500m_up"), (5000.0, -20.0, "5km_up") };
                var lodFrames = new Dictionary<string, Color32[]>();
                foreach ((double meters, double pitch, string tag) in series)
                {
                    Pose(at, meters, facing, pitch);
                    SettleRivers();
                    Color32[] img = Render();
                    lodFrames[tag] = img;
                    RiverSurfaceDiagnostics d = visual.RiverSurface.Diagnostics;
                    Save(img, $"B_lod_{tag}", "G.2D LOD", $"gorge descent/ascent {tag}: levels far {d.FarPieces} mid {d.MidPieces} near {d.NearPieces}, samples {d.DisplayedSamples}, cache {d.CachedPieces} pieces {d.CacheBytes / 1048576.0:0.0} MB");
                }
                log.Append(string.Format(Inv, "\nLOD descent / ascent at the gorge: same pose 5 km down vs up differs on {0:0.00} % of pixels, 500 m down vs up {1:0.00} %",
                    Changed(lodFrames["5km"], lodFrames["5km_up"]), Changed(lodFrames["500m"], lodFrames["500m_up"])));
            }

            // ---- Lake interaction (B, lakes shown): a reach entering a lake, if any ribbon has one.
            {
                visual.Lakes.showLakes = true;
                visual.RefreshColors();
                RiverRibbonBuilder sp = visual.RiverRibbons;
                int lakeSample = -1;
                for (int i = 0; i < sp.SpineSampleCount && lakeSample < 0; i++)
                {
                    RiverSpineSample s = sp.GetSpineSample(i);
                    if (s.ClampToLake > 0 && s.WaterRadius <= 0f && Math.Abs(s.Direction.y) < 0.75) lakeSample = i;
                }
                if (lakeSample >= 0)
                {
                    Vector3d at = new Vector3d(sp.GetSpineSample(lakeSample).Direction).Normalized;
                    LightPoint(at, 0.55f);
                    foreach ((string config, int placement) in configs)
                    {
                        visual.Rivers.placement = placement;
                        visual.RefreshColors();
                        foreach ((double meters, double pitch, string tag) in new[] { (20000.0, -35.0, "20km"), (2000.0, -15.0, "2km") })
                        {
                            Pose(at, meters, PlanetTangentFrame.At(at).North, pitch);
                            SettleRivers();
                            Save(Render(), $"{config}_lake_{tag}_pbr", "G.2D " + config, $"lake transition sample {lakeSample}, {tag}, placement {(placement == 1 ? "canonical" : "legacy")}");
                        }
                    }
                    log.Append(string.Format(Inv, "\nLake interaction: spine sample {0} at ({1:0.000000}, {2:0.000000}, {3:0.000000})", lakeSample, at.x, at.y, at.z));
                }
                else
                {
                    log.Append("\nLake interaction: no ribbon with a rendered-shoreline transition on this planet (lakes shown)");
                }
                visual.Lakes.showLakes = false;
            }

            // ---- Plateau / plain regression (seed 310929, rivers shown): both placements, 2 km.
            {
                var regression = new[]
                {
                    ("plateau", new Vector3d(0.131565, 0.508858, -0.850737), new Vector3d(-0.894736, 0.430425, 0.119085)),
                    ("plain", new Vector3d(-0.195978, 0.181058, 0.963748), new Vector3d(0.727489, -0.632164, 0.266699)),
                };
                host.Definition.seed = 310929;
                host.Regenerate();
                planet = host.Current;
                explorer.Planet = planet;
                foreach ((string config, int placement) in configs)
                {
                    visual.Rivers.placement = placement;
                    visual.RefreshColors();
                    foreach ((string name, Vector3d at0, Vector3d facing0) in regression)
                    {
                        Vector3d at = at0.Normalized;
                        LightPoint(at, 0.55f);
                        Pose(at, 2000.0, (facing0 - at * Vector3d.Dot(facing0, at)).Normalized, -12.0);
                        SettleRivers();
                        Color32[] img = Render();
                        frames[$"{config}_{name}"] = img;
                        Save(img, $"{config}_{name}_2km_pbr", "G.2D " + config, $"seed 310929 {name}, 2 km, placement {(placement == 1 ? "canonical" : "legacy")}");
                        if (config == "B") log.Append(string.Format(Inv, "\n{0} (seed 310929, 2 km): A vs B changed pixels {1:0.0} %", name, Changed(frames["A_" + name], img)));
                    }
                }
            }

            File.WriteAllText(Path.Combine(Out, "captures.md"), "# V0.2.0G.2D A / B captures (batch, EditMode rendering)\n\n" + meta);
            File.WriteAllText(Path.Combine(Out, "measurements.txt"), log.ToString().Trim() + "\n");
            Debug.Log("VALIDATION" + log);
            if (target != null) Object.DestroyImmediate(target);
            if (readback != null) Object.DestroyImmediate(readback);
        }
    }
